import java.util.*;

public class Solution {

public static void main(String[] args) {
Scanner sc = new Scanner(System.in);

int n = sc.nextInt();
int k = sc.nextInt();

int[] arr = new int[n];

for (int i = 0; i < n; i++) {
arr[i] = sc.nextInt();
}

Deque<Integer> deque = new ArrayDeque<>();
Map<Integer, Integer> frequency = new HashMap<>();

int unique = 0;
int maxUnique = 0;

for (int i = 0; i < n; i++) {

// Add current element
deque.addLast(arr[i]);

frequency.put(arr[i], frequency.getOrDefault(arr[i], 0) + 1);

if (frequency.get(arr[i]) == 1) {
unique++;
}

// If window size becomes greater than k
if (deque.size() > k) {
int removed = deque.removeFirst();

frequency.put(removed, frequency.get(removed) - 1);

if (frequency.get(removed) == 0) {
unique--;
}
}

// Check maximum unique values
if (deque.size() == k) {
maxUnique = Math.max(maxUnique, unique);
}
}

System.out.println(maxUnique);

sc.close();
}
}
